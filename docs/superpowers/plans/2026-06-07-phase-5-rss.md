```python
"""Read-only SQLite reader for zotero.sqlite (design §11)."""
from __future__ import annotations
import sqlite3
from pathlib import Path
from typing import Self

from zotero_cli.models.errors import (
    SqliteNotFoundError, SqliteLockedError, SqliteSchemaIncompatibleError,
)

_REQUIRED_TABLES = {
    "feeds", "feedItems", "items", "itemData", "itemDataValues", "fields",
    "itemCreators", "creators", "creatorTypes",
}
_REQUIRED_FEEDITEMS_COLS = {"itemID", "guid", "readTime", "translatedTime"}


class SQLiteReader:
    def __init__(self, conn: sqlite3.Connection, path: Path) -> None:
        self.conn = conn
        self.path = path

    @classmethod
    def open(cls, path: Path) -> Self:
        if not path.exists():
            raise SqliteNotFoundError(
                f"SQLite database not found: {path}",
                hint="Set [<profile>.sqlite] path or ZOTERO_DATA_DIR env var.",
            )
        uri = f"file:{path}?mode=ro&nolock=1"
        try:
            conn = sqlite3.connect(uri, uri=True, timeout=5.0)
        except sqlite3.OperationalError as e:
            raise SqliteLockedError(f"Cannot open SQLite: {e}") from e
        cls._check_schema(conn)
        return cls(conn, path)

    @staticmethod
    def _check_schema(conn: sqlite3.Connection) -> None:
        rows = conn.execute(
            "SELECT name FROM sqlite_master WHERE type='table'"
        ).fetchall()
        present = {r[0] for r in rows}
        missing = _REQUIRED_TABLES - present
        if missing:
            raise SqliteSchemaIncompatibleError(
                f"Missing required tables: {sorted(missing)}",
                hint="Verify zotero.sqlite is from Zotero 6+ and not corrupted.",
            )
        cols = {r[1] for r in conn.execute("PRAGMA table_info(feedItems)").fetchall()}
        col_missing = _REQUIRED_FEEDITEMS_COLS - cols
        if col_missing:
            raise SqliteSchemaIncompatibleError(
                f"feedItems missing columns: {sorted(col_missing)}"
            )

    def close(self) -> None:
        self.conn.close()
```

```python
_REQUIRED_FIELDS = ("title", "date", "url", "abstractNote")


@classmethod
def open(cls, path: Path) -> Self:
    # ... 上一任务的逻辑 ...
    cls._check_schema(conn)
    field_ids = cls._load_field_ids(conn)
    return cls(conn=conn, path=path, field_ids=field_ids)


@staticmethod
def _load_field_ids(conn: sqlite3.Connection) -> dict[str, int]:
    rows = conn.execute(
        "SELECT fieldID, fieldName FROM fields WHERE fieldName IN (?,?,?,?)",
        _REQUIRED_FIELDS,
    ).fetchall()
    mapping = {name: fid for fid, name in rows}
    missing = set(_REQUIRED_FIELDS) - mapping.keys()
    if missing:
        raise SqliteSchemaIncompatibleError(
            f"Missing required fields in fields table: {sorted(missing)}"
        )
    return mapping
```

## Task 5: adapters/sqlite_reader.py — 异常翻译

**Files:** Modify `sqlite_reader.py` 与测试

**关键测试**：

```python
        side_effect=sqlite3.OperationalError("database is locked"),
    )

    with pytest.raises(SqliteLockedError):

```

**实现**（细化 `open()` 的 try/except）：

```python
@classmethod
def open(cls, path: Path) -> Self:
    if not path.exists():
        raise SqliteNotFoundError(f"SQLite database not found: {path}")
    uri = f"file:{path}?mode=ro&nolock=1"
    try:
        conn = sqlite3.connect(uri, uri=True, timeout=5.0)
        # 触发实际打开：sqlite3.connect 是惰性的，先做一次 PRAGMA
        conn.execute("PRAGMA schema_version")
    except sqlite3.OperationalError as e:
        msg = str(e).lower()
        if "lock" in msg:
            raise SqliteLockedError(f"SQLite database is locked: {path}") from e
        raise SqliteSchemaIncompatibleError(
            f"Cannot open SQLite: {e}",
            hint="Verify file is a valid Zotero SQLite database.",
        ) from e
    except sqlite3.DatabaseError as e:
        raise SqliteSchemaIncompatibleError(f"Invalid SQLite file: {e}") from e
    cls._check_schema(conn)
    field_ids = cls._load_field_ids(conn)
    return cls(conn=conn, path=path, field_ids=field_ids)
```

```python
"""Feed query service (design §11.3)."""
from __future__ import annotations
from zotero_cli.adapters.sqlite_reader import SQLiteReader
from zotero_cli.models.feed import FeedSummary
from zotero_cli.models.errors import FeedNotFoundError

_LIST_FEEDS_SQL = """
SELECT
    f.libraryID,
    f.name,
    f.url,
    f.lastUpdate,
    f.lastCheck,
    f.lastCheckError,
    f.refreshInterval,
    COUNT(fi.itemID) AS total_count,
    SUM(CASE WHEN fi.readTime IS NULL THEN 1 ELSE 0 END) AS unread_count
FROM feeds f
LEFT JOIN items i ON i.libraryID = f.libraryID
LEFT JOIN feedItems fi ON fi.itemID = i.itemID
GROUP BY f.libraryID
ORDER BY f.name;
"""


class FeedService:
    def __init__(self, reader: SQLiteReader) -> None:
        self._reader = reader

    def list_feeds(self) -> list[FeedSummary]:
        rows = self._reader.conn.execute(_LIST_FEEDS_SQL).fetchall()
        return [
            FeedSummary(
                feed_id=r[0], name=r[1], url=r[2],
                last_update=r[3], last_check=r[4], last_check_error=r[5],
                refresh_interval=r[6], total_count=r[7] or 0,
                unread_count=int(r[8] or 0),
            )
            for r in rows
        ]

    def get_feed(self, feed_id: int) -> FeedSummary:
        match = next((f for f in self.list_feeds() if f.feed_id == feed_id), None)
        if match is None:
            raise FeedNotFoundError(
                f"Feed with id {feed_id} not found",
                hint="Use 'zotero-cli feeds list' to see available feeds.",
            )
        return match
```

```python
from zotero_cli.utils.date_parser import DateRange
from zotero_cli.models.feed import FeedItem

_LIST_ITEMS_SQL = """
SELECT
    fi.itemID, fi.guid, fi.readTime, fi.translatedTime,
    i.dateAdded, i.dateModified,
    title_v.value     AS title,
    date_v.value      AS date_raw,
    SUBSTR(date_v.value, 1, 10) AS date_sql,
    url_v.value       AS url,
    abstract_v.value  AS abstract
FROM feedItems fi
JOIN items i ON i.itemID = fi.itemID
LEFT JOIN itemData title_id ON title_id.itemID = i.itemID AND title_id.fieldID = ?
LEFT JOIN itemDataValues title_v ON title_v.valueID = title_id.valueID
LEFT JOIN itemData date_id ON date_id.itemID = i.itemID AND date_id.fieldID = ?
LEFT JOIN itemDataValues date_v ON date_v.valueID = date_id.valueID
LEFT JOIN itemData url_id ON url_id.itemID = i.itemID AND url_id.fieldID = ?
LEFT JOIN itemDataValues url_v ON url_v.valueID = url_id.valueID
LEFT JOIN itemData abstract_id ON abstract_id.itemID = i.itemID AND abstract_id.fieldID = ?
LEFT JOIN itemDataValues abstract_v ON abstract_v.valueID = abstract_id.valueID
WHERE i.libraryID = ?
  AND (
    (? = 1 AND date_v.value IS NULL)
    OR (
      date_v.value IS NOT NULL
      AND SUBSTR(date_v.value, 1, 10) >= ?
      AND SUBSTR(date_v.value, 1, 10) <= ?
    )
  )
ORDER BY SUBSTR(date_v.value, 1, 10) DESC NULLS LAST
LIMIT ?;
"""


def list_items(
    self,
    feed_id: int,
    *,
    date_filter: DateRange | None = None,
    include_undated: bool = False,
    limit: int = 100,
) -> list[FeedItem]:
    fids = self._reader.field_ids
    rng = date_filter or DateRange("0000-00-00", "9999-12-31")
    params = (
        fids["title"], fids["date"], fids["url"], fids["abstractNote"],
        feed_id,
        1 if include_undated else 0,
        rng.start, rng.end,
        limit,
    )
    rows = self._reader.conn.execute(_LIST_ITEMS_SQL, params).fetchall()
    return [
        FeedItem(
            feed_id=feed_id, item_id=r[0], guid=r[1],
            read_time=r[2], translated_time=r[3],
            date_added=r[4], date_modified=r[5],
            title=r[6], date_raw=r[7], date_sql=r[8],
            url=r[9], abstract=r[10], creators=[],
        )
        for r in rows
    ]
```

```python
from zotero_cli.models.feed import FeedItemCreator

_CREATORS_SQL_TPL = """
SELECT ic.itemID, c.firstName, c.lastName, c.fieldMode,
       ct.creatorType, ic.orderIndex
FROM itemCreators ic
JOIN creators c ON c.creatorID = ic.creatorID
JOIN creatorTypes ct ON ct.creatorTypeID = ic.creatorTypeID
WHERE ic.itemID IN ({placeholders})
ORDER BY ic.itemID, ic.orderIndex;
"""


def _load_creators(self, item_ids: list[int]) -> dict[int, list[FeedItemCreator]]:
    if not item_ids:
        return {}
    placeholders = ",".join("?" * len(item_ids))
    sql = _CREATORS_SQL_TPL.format(placeholders=placeholders)
    rows = self._reader.conn.execute(sql, item_ids).fetchall()
    by_item: dict[int, list[FeedItemCreator]] = {iid: [] for iid in item_ids}
    for iid, first, last, mode, ctype, oidx in rows:
        by_item[iid].append(FeedItemCreator(
            first_name=first, last_name=last, field_mode=mode,
            creator_type=ctype, order_index=oidx,
        ))
    return by_item


# 在 list_items 末尾构造 FeedItem 之前先取 creators：
items_no_creators = [...]  # 现有逻辑
creators_by_id = self._load_creators([i.item_id for i in items_no_creators])
for item in items_no_creators:
    item.creators = creators_by_id.get(item.item_id, [])
return items_no_creators
```

```python
class FeedItemCreator(BaseModel):

    first_name: str | None
    last_name: str | None
    field_mode: int  # 0 = person, 1 = single-field (organization)
    creator_type: str
    order_index: int


class FeedItem(BaseModel):
    """Per Phase 1 接口契约 §B：list 项必须有 `key` 字段供 --quiet 输出消费。"""
    model_config = ConfigDict(frozen=False)
    feed_id: int        # = items.libraryID, 等于 feeds.libraryID（见设计 §11.2）
    item_id: int
    guid: str | None
    read_time: str | None
    translated_time: str | None
    date_added: str | None
    date_modified: str | None
    title: str | None
    date_raw: str | None
    date_sql: str | None  # 前 10 字符 'YYYY-MM-DD'，可为占位 '2024-00-00'
    url: str | None
    abstract: str | None
    creators: list[FeedItemCreator]

    @computed_field  # type: ignore[misc]
    @property
    def key(self) -> str:
        return str(self.item_id)

    last_check: str | None
```

**实现骨架**（**全部经 `run_command`**——本模块不写 try/except CLIError，不直接 `typer.echo` 写 stderr）：

```python
"""feeds command tree (design §6 / §11)."""
from __future__ import annotations
import typer
from zotero_cli.adapters.sqlite_reader import SQLiteReader
from zotero_cli.services.config_service import load_config
from zotero_cli.services.feed_service import FeedService
from zotero_cli.commands._runner import GlobalOptions, run_command
from zotero_cli.utils.date_parser import date_range_to_sql_bounds
from zotero_cli.utils.output import OutputMode

app = typer.Typer(no_args_is_help=True, help="RSS feed read-only queries")


def _open_reader(options: GlobalOptions) -> SQLiteReader:
    cfg = load_config(profile=options.profile, config_path=options.config_path)
    return SQLiteReader.open(cfg.sqlite.path)


@app.command("list")
def cmd_list(ctx: typer.Context) -> None:
    options: GlobalOptions = ctx.obj

    def work():
        reader = _open_reader(options)
        try:
            feeds = FeedService(reader).list_feeds()
        finally:
            reader.close()
        return [f.model_dump() for f in feeds]

    run_command(command="feeds.list", mode=OutputMode.KV_LIST,
                options=options, work=work)


@app.command("show")
def cmd_show(ctx: typer.Context, feed_id: int) -> None:
    options: GlobalOptions = ctx.obj

    def work():
        reader = _open_reader(options)
        try:
            feed = FeedService(reader).get_feed(feed_id)  # 抛 FeedNotFoundError
        finally:
            reader.close()
        return feed.model_dump()

    run_command(command="feeds.show", mode=OutputMode.KV,
                options=options, work=work)


@app.command("items")
def cmd_items(
    ctx: typer.Context,
    feed_id: int,
    date: str | None = typer.Option(None, "--date"),
    include_undated: bool = typer.Option(False, "--include-undated"),
    limit: int = typer.Option(100, "--limit"),
    all_fields: bool = typer.Option(False, "--all-fields"),
) -> None:
    options: GlobalOptions = ctx.obj
    cfg = load_config(profile=options.profile, config_path=options.config_path)
    field_filter = None if all_fields else cfg.feed_item_fields.list

    def work():
        reader = _open_reader(options)
        try:
            svc = FeedService(reader)
            svc.get_feed(feed_id)  # FEED_NOT_FOUND 边界
            date_range = date_range_to_sql_bounds(date) if date else None
            items = svc.list_items(
                feed_id=feed_id, date_filter=date_range,
                include_undated=include_undated, limit=limit,
            )
        finally:
            reader.close()
        return [i.model_dump() for i in items]

    run_command(command="feeds.items", mode=OutputMode.KV_LIST,
                options=options, work=work, field_filter=field_filter)
```

> **回应 review P1 Issue 2 + 3**：本模块 `ctx.obj` 一律按属性访问（`options.profile` / `options.json_mode` / `options.quiet`），与 Phase 3 / 6 保持一致。错误处理与 stdout/stderr 分离全部委托给 `run_command`。

> **字段过滤与 `--all-fields`**：`[<profile>.feed_item_fields] list` 决定默认字段；`--all-fields` 覆盖（设计 §7.4）。本任务通过 `run_command(field_filter=...)` 传给 Phase 1 的 `render`。

> **`--quiet` 输出键名**：`render()` 取每项 `key` 字段。FeedSummary / FeedItem 已在 Task 9 通过 `computed_field` 暴露 `key`（值为 `str(feed_id)` / `str(item_id)`），`--quiet` 模式下 stdout 自动得到 ID 列表，每行一个。命令层不需要额外处理。
